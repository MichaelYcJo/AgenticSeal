"""#1236 phase 4: the edition pairs E1 to E4, read from the copy "$S/tree4"
of 94635718e that the setup block extracts. It cuts each edition into its
sections, at each live heading, and each section into the blocks a reader
compares: a paragraph, a list item, a table row other than the delimiter
row, and a fenced block whole. It pairs the two editions' sections in order,
and within a section their blocks in order, and prints each pair for the
reading design M7 makes. It decides no meaning (design G2). A heading is
`tests/behaviour/test_both_editions_carry_the_same_folds.py#HEADING` outside
a fence, the outline that module holds (design G13).
"""

import os
import re
import sys

T = os.path.join(os.environ["S"], "tree4")
PAIRS = {
    "E1": ("README.md", "README.ko.md"),
    "E2": ("docs/layout/one-root-by-lifetime.md", "docs/layout/one-root-by-lifetime.ko.md"),
    "E3": ("docs/experiments/README.md", "docs/experiments/README.ko.md"),
    "E4": ("docs/experiments/2026-09-03-skill-preload-and-the-copy-in-force.md",
           "docs/experiments/2026-09-03-skill-preload-and-the-copy-in-force.ko.md"),
}
HEADING = re.compile(r"^ {0,3}(#{1,6})(?:[ \t]|$)")
FENCE = re.compile(r"^ {0,3}(```|~~~)")
ITEM = re.compile(r"^ {0,3}(?:[-*]|\d+\.)\s")
DELIMITER = re.compile(r"^\|[-:| ]+\|\s*$")


def sections(path):
    """`[(line, heading, [(line, block)])]`: the text before the first
    heading is a section whose heading is empty."""
    out = [(0, "", [])]
    cur, fence = None, None
    for k, line in enumerate(open(os.path.join(T, path), encoding="utf-8").read().split("\n"), 1):
        if fence:
            cur[1].append(line)
            if line.strip().startswith(fence):
                fence, cur = None, None
            continue
        if FENCE.match(line):
            fence = FENCE.match(line).group(1)
            cur = (k, [line])
            out[-1][2].append(cur)
            continue
        if HEADING.match(line):
            out.append((k, line.strip(), []))
            cur = None
            continue
        if not line.strip():
            cur = None
            continue
        if line.startswith("|"):
            if not DELIMITER.match(line):
                out[-1][2].append((k, [line]))
            cur = None
            continue
        if cur is None or ITEM.match(line):
            cur = (k, [line])
            out[-1][2].append(cur)
        else:
            cur[1].append(line)
    return out


def pair(label, verbose):
    en, ko = (sections(p) for p in PAIRS[label])
    differ = 0
    for i in range(max(len(en), len(ko))):
        a = en[i] if i < len(en) else (0, "-", [])
        b = ko[i] if i < len(ko) else (0, "-", [])
        same = len(a[2]) == len(b[2])
        differ += not same
        print(f"{label} section {i}: {a[1] or '(before the first heading)'} | {b[1] or '(before the first heading)'}"
              f" | blocks {len(a[2])} {len(b[2])}{'' if same else ' DIFFER'}")
        if verbose:
            for j in range(max(len(a[2]), len(b[2]))):
                x = a[2][j] if j < len(a[2]) else (0, ["-"])
                y = b[2][j] if j < len(b[2]) else (0, ["-"])
                print(f"    {j}: en:{x[0]} {' '.join(x[1])[:70]!r}\n       ko:{y[0]} {' '.join(y[1])[:70]!r}")
    n_en, n_ko = (sum(len(s[2]) for s in e) for e in (en, ko))
    print(f"{label} sections {len(en)} {len(ko)}; blocks {n_en} {n_ko}; sections whose block counts differ {differ}")


if __name__ == "__main__":
    for label in [a for a in sys.argv[1:] if a != "-v"] or PAIRS:
        pair(label, "-v" in sys.argv)
EOF
```
