def is_python(p, t):
    return p.endswith(".py") or t.startswith("#!/usr/bin/env python")


def segments(p, constants=False):
    """`[[(line, text)]]`: the raw prose of the code file P. For Python each
    docstring, each run of comment lines on consecutive lines, and, where
    CONSTANTS, each other string constant or f-string (rendered with `{}`)
    of four words or more; for any other file each run of comment lines."""
    t = P.text_of(p)
    if t is None:
        return []
    out = []
    if is_python(p, t):
        try:
            tree = ast.parse(t)
        except SyntaxError:
            return []
        docs, inner = set(), set()
        for n in ast.walk(tree):
            if isinstance(n, (ast.Module, ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                b = n.body
                if b and isinstance(b[0], ast.Expr) and isinstance(b[0].value, ast.Constant) \
                        and isinstance(b[0].value.value, str):
                    docs.add(id(b[0].value))
                    out.append([(b[0].value.lineno + k, x) for k, x in enumerate(b[0].value.value.split("\n"))])
            if isinstance(n, ast.JoinedStr):
                inner |= {id(v) for v in n.values}
        if constants:
            for n in ast.walk(tree):
                if id(n) in docs or id(n) in inner:
                    continue
                s = P2.render(n)
                if s is not None and len(s.split()) >= 4:
                    out.append([(n.lineno + k, x) for k, x in enumerate(s.split("\n"))])
        block, last = [], None
        for tok in tokenize.generate_tokens(io.StringIO(t).readline):
            if tok.type == tokenize.COMMENT:
                if last is not None and tok.start[0] != last + 1 and block:
                    out.append(block)
                    block = []
                block.append((tok.start[0], tok.string.lstrip("#")))
                last = tok.start[0]
        if block:
            out.append(block)
        return out
    block = []
    for k, line in enumerate(t.split("\n"), 1):
        m = re.match(r"^\s*(#|::|rem\b|REM\b)(.*)", line)
        if m:
            block.append((k, m.group(2)))
        elif block:
            out.append(block)
            block = []
    if block:
        out.append(block)
    return out

def blocks(segs):
    """Each paragraph of each segment: a run of lines with text."""
    out = []
    for seg in segs:
        cur = []

                cur.append((k, x))

                cur = []
        if cur:
            out.append(cur)
    return out


def coordinates(label):
    """Each `path:line` and `path#name` coordinate in the row's segments whose
    path names a file of the tree from its top. One past the file's end, or
    whose name no line of the file defines or assigns, is marked BAD; one
    that resolves is marked OK, and each is read before it counts."""
    line_re = re.compile(r"(?<![\w./-])((?:[\w-]+/)*[\w.-]+\.(?:py|md|yml|yaml|json|sh|toml)):(\d+)(?:-(\d+))?")
    name_re = re.compile(r"(?<![\w./-])((?:[\w-]+/)*[\w.-]+\.py)#(\w+)")
    exists = set(P.ALL)
    n_line = n_name = bad = 0
    for p, segs in sorted(corpus(label).items()):
        for seg in segs:
            for k, line in seg:
                for m in line_re.finditer(line):
                    path, a = m.group(1), int(m.group(2))
                    if path not in exists:
                        continue
                    n_line += 1
                    length = len(P.read(path).split("\n"))
                    ok = a <= length and not (m.group(3) and int(m.group(3)) > length)
                    bad += not ok
                    print(f"{'OK ' if ok else 'BAD'} line {p}:{k} -> {m.group(0)} ({length} lines)")
                for m in name_re.finditer(line):
                    path, name = m.group(1), m.group(2)
                    if path not in exists:
                        continue
                    n_name += 1
                    ok = bool(re.search(rf"^\s*(?:async\s+)?(?:def|class)\s+{name}\b|^\s*{name}\s*[:=]|^\s*{name}\s*,|,\s*{name}\s*=",
                                        P.read(path), re.M))
                    bad += not ok
                    if not ok or "-v" in sys.argv:
                        print(f"{'OK ' if ok else 'BAD'} name {p}:{k} -> {m.group(0)}")
    print(f"{label} path:line coordinates {n_line}; path#name coordinates {n_name}; not resolving {bad}")

def lists(label):
    """Each block of the row's segments that names at least three members of
    an enumerable source and at least half of them, against that source. A

def defined(p):
    """Every name the module P defines or binds at any depth: functions,
    classes, `Class::method`, and names assigned."""
    tree = ast.parse(P.read(p))
    out = set()
    for n in ast.walk(tree):
        if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            out.add(n.name)
            if isinstance(n, ast.ClassDef):
                out |= {f"{n.name}::{m.name}" for m in n.body if isinstance(m, (ast.FunctionDef, ast.AsyncFunctionDef))}
        if isinstance(n, (ast.Assign, ast.AnnAssign)):
            for tg in (n.targets if isinstance(n, ast.Assign) else [n.target]):
                out |= {x.id for x in ast.walk(tg) if isinstance(x, ast.Name)}
    return out


CASE = re.compile(r"^(?:(?P<path>[\w./-]+\.py)(?:::|#))?(?:(?P<cls>\w+)::)?(?P<name>test_\w+)(?:\[[^\]]*\])?(?:\(\))?$")
HASH = re.compile(r"^(?P<path>[\w./-]+\.py)?#(?P<name>[A-Za-z_]\w*)(?:\(\))?$")
MODULE = re.compile(r"^(?:[\w./-]+/)?test_\w+\.py$")


def enumerations():
    """Design M5's docstring enumeration, R19: each backticked span in a
    docstring or comment of a test module that names a case (`test_*`,
    `Class::test_*`, `file.py::test_*`), a module (`test_*.py`) or a name
    (`#name`, `file.py#name`), against the names the module it points at
    defines, read from its AST. A span whose path names a file from the
    tree's top is a coordinate, `coordinates`' share, and is not read
    here. A bare name is looked for in its own module first, then in every
    test module. With `--module`, module docstrings alone, the reading
    #506's design states in its *Grounding*."""
    module_only = "--module" in sys.argv
    mods = [p for p in P.PARTS["P9"] if p.endswith(".py")]
    names = {p: defined(p) for p in mods}
    exists = set(P.ALL)

    def resolve(path):
        if path in exists:
            return path
        hits = [q for q in exists if q.endswith("/" + path)]
        return hits[0] if len(hits) == 1 else None

    tally = collections.Counter()
    for p in mods:
        segs = segments(p)
        if module_only:
            segs = segs[:1] if ast.get_docstring(ast.parse(P.read(p))) else []
        for seg in segs:
            for k, line in seg:
                for span in re.findall(r"`([^`\n]+)`", line):
                    span = span.strip()
                    if MODULE.match(span):
                        where = span if span in exists else resolve(span)
                        verdict = f"module {where}" if where else "NO SUCH MODULE"
                    elif (m := CASE.match(span)) or (m := HASH.match(span)):
                        path, name = m.group("path"), m.group("name")
                        key = f"{m.group('cls')}::{name}" if "cls" in m.groupdict() and m.group("cls") else name
                        if path and path in exists and span.startswith(path + "#"):
                            continue
                        if path:
                            target = resolve(path)
                            if target is None or not target.endswith(".py"):
                                verdict = "PATH NOT RESOLVED"
                            elif key in defined(target):
                                verdict = f"in {target}"
                            else:
                                verdict = f"NOT IN {target}"
                        elif key in names[p]:
                            verdict = "own module"
                        else:
                            elsewhere = sorted(q for q in mods if key in names[q])
                            verdict = f"ELSEWHERE {', '.join(elsewhere[:3])}" if elsewhere else "NOWHERE"
                    else:
                        continue
                    tally[verdict.split()[0]] += 1
                    if verdict.split()[0].isupper() or "-v" in sys.argv:
                        print(f"{verdict} :: {p}:{k} `{span}`")
    print(f"R19 {'module docstring' if module_only else 'docstring and comment'} spans {sum(tally.values())}; "
          + "; ".join(f"{k} {v}" for k, v in sorted(tally.items())))

def gh_issue(n, cache={}):
    if n not in cache:
        r = subprocess.run(["gh", "issue", "view", str(n), "--json", "number,state,milestone,title"],
                           capture_output=True, text=True)
        cache[n] = json.loads(r.stdout) if r.returncode == 0 else None
    return cache[n]


def tracker():
    """Design M6, R18, over "$S/issues.json" (the open issues with bodies and
