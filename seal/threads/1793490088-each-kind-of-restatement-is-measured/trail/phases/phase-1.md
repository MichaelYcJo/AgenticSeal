def reader():
    ns = {"re": re, "itertools": itertools}
    keep = {"WINDOW", "FENCE", "HEADING", "SECTION", "SECTION_TOKEN", "STRIP", "lines",
            "words", "windows", "shared_counts", "BASELINE", "TOP_FILES", "TOP_DIRS"}
    for rel in ("tests/text/test_a_moved_rule_leaves_its_definition.py",
                "tests/text/test_no_passage_is_pasted_into_a_second_file.py"):
        body = [n for n in ast.parse(read(rel)).body
                if (isinstance(n, ast.Assign) and isinstance(n.targets[0], ast.Name)
                    and n.targets[0].id in keep)
                or (isinstance(n, ast.FunctionDef) and n.name in keep)]
        exec(compile(ast.Module(body, []), rel, "exec"), ns)
    return ns

def passages(kind):
    """Each maximal run of at least WINDOW words a pair of the kind shares; a
    run another run contains is the same passage, with the holders of both."""
    ns = reader()
    W, words, windows = ns["WINDOW"], ns["words"], ns["windows"]
    text = {}
    for p in ALL:
        try:
            text[p] = words(read(p))
        except UnicodeDecodeError:
            pass
    pairs = {tuple(sorted((a, b))) for x, y in KINDS[kind]
             for a in PARTS[x] for b in PARTS[y] if a != b and a in text and b in text}
    found = {}
    for a, b in sorted(pairs):
        wa = text[a]
        common = windows(wa) & windows(text[b])
        start = None
        for i in range(len(wa) - W + 2):
            inside = bool(common) and i < len(wa) - W + 1 and " ".join(wa[i:i + W]) in common
            if inside and start is None:
                start = i
            if not inside and start is not None:
                found.setdefault(" ".join(wa[start:i - 1 + W]), set()).update((a, b))
                start = None
    for run in sorted(found, key=len):
        longer = [r for r in found if r != run and run in r]
        if longer:
            found[max(longer, key=len)].update(found.pop(run))
    rows = sorted(found.items(), key=lambda kv: (sorted(kv[1]), kv[0]))
    for n, (run, holders) in enumerate(rows, 1):
        print(f"{kind}.{n} [{len(run.split())} words] {' + '.join(sorted(holders))}")
        print(f"    {run[:300]}")
    print(f"{kind}: {len(found)} passages over {len(pairs)} file pairs")

def editions():
    """Each `X.ko.md` beside `X.md`: the outline, and per section by position
    the backticked tokens and the figures one edition holds and the other not."""
    head = re.compile(r"^ {0,3}(#{1,6})(?:[ \t]|$)(.*)")
    fence_re = re.compile(r"^ *(`{3,}|~{3,})")
    tick = re.compile(r"`([^`\n]+)`")
    fig = re.compile(r"(?<![\w.#/-])\d+(?:[.,]\d+)?(?![\w/-])")

    def sections(rel):
        out, cur, fence = [], ["(top)", []], None
        for line in read(rel).split("\n"):
            m = fence_re.match(line)
            if fence or m:
                if fence and m and m.group(1)[0] == fence[0]:
                    fence = None
                elif not fence:
                    fence = m.group(1)
                cur[1].append(line)
                continue
            h = head.match(line)
            if h:
                out.append(cur)
                cur = [f"{h.group(1)} {h.group(2).strip()}", []]
            else:
                cur[1].append(line)
        return out + [cur]

    def facts(body):
        text = "\n".join(body)
        return set(tick.findall(text)), set(fig.findall(tick.sub(" ", text)))

    total = 0
    for name in sorted(p[:-6] for p in ALL if p.endswith(".ko.md")):
        en, ko = sections(name + ".md"), sections(name + ".ko.md")
        same = [h.split(" ")[0] for h, _ in en] == [h.split(" ")[0] for h, _ in ko]
        print(f"== {name}: {len(en)} and {len(ko)} sections; outline {'same' if same else 'DIFFERS'}")
        for i, ((he, be), (hk, bk)) in enumerate(zip(en, ko)):
            (te, fe), (tk, fk) = facts(be), facts(bk)
            if te ^ tk or fe ^ fk:
                total += 1
                print(f"  [{i}] {he[:60]} | {hk[:40]}")
                for label, xs in (("en only", (te - tk) | (fe - fk)), ("ko only", (tk - te) | (fk - fe))):
                    if xs:
                        print(f"      {label}: {sorted(xs)}")
    print(f"sections whose tokens or figures differ: {total}")


def pins(verbose):
    """An assert under tests/ whose test is a string literal `in` an
    expression, the literal at least four words, standing whitespace-folded
    in a file of docs/, skills/ or agents/."""
    rules = {p: " ".join(read(p).split()) for p in PARTS["P1"] + PARTS["P2"] + PARTS["P3"]}

    def literal(node):
        if isinstance(node, ast.Constant) and isinstance(node.value, str):
            return node.value
        if isinstance(node, ast.JoinedStr):
            parts = [v.value for v in node.values if isinstance(v, ast.Constant)]
            return max(parts, key=len) if parts else None
        return None

    per_module, rows = collections.Counter(), []
    for rel in [p for p in ALL if p.startswith("tests/") and p.endswith(".py")]:
        for a in ast.walk(ast.parse(read(rel))):
            if not isinstance(a, ast.Assert):
                continue
            for c in ast.walk(a.test):
                if isinstance(c, ast.Compare) and len(c.ops) == 1 and isinstance(c.ops[0], ast.In):
                    lit = literal(c.left)
                    folded = " ".join(lit.split()) if lit else ""
                    if len(folded.split()) >= 4:
                        homes = [r for r, t in rules.items() if folded in t]
                        if homes:
                            per_module[rel] += 1
                            rows.append((rel, a.lineno, folded, homes))
    if verbose:
        for rel, line, folded, homes in rows:
            print(f"{rel}:{line} {folded[:80]!r} -> {', '.join(homes[:2])}")
    for rel, k in per_module.most_common():
        print(f"{k:4} {rel}")
    print(f"pins {len(rows)} in {len(per_module)} modules")


def templates():
    """A backticked `## Heading` in a paragraph of docs/, skills/ or agents/
    that names a template, against that template's headings; a backticked
    table row in those files, against every template's text."""
    heads = {p: set(re.findall(r"(?m)^(#{1,6} .+?)\s*$", read(p))) for p in PARTS["P4"] if p.endswith(".md")}
    flat = [" ".join(read(p).split()) for p in PARTS["P4"]]
    tok, tp = re.compile(r"`(#{2,3} [^`\n]+)`"), re.compile(r"templates/(?:records|setup)/[\w.-]+\.md")
    listed, unmatched, rows, rows_out = 0, [], 0, []
    for p in PARTS["P1"] + PARTS["P2"] + PARTS["P3"]:
        text = read(p)
        for para in re.split(r"\n\s*\n", text):
            names = set(tp.findall(para))
            for t in tok.findall(para) if names else ():
                listed += 1
                if not any(t.strip() in heads.get(n, ()) for n in names):
                    unmatched.append(f"{p}: {t.strip()} (names {', '.join(sorted(names))})")
        for r in re.findall(r"`(\|[^`\n]*\|)`", text):
            rows += 1
            if not any(" ".join(r.split()) in f for f in flat):
                rows_out.append(f"{p}: {r}")
    print(f"heading tokens in a paragraph naming a template: {listed}; not a heading of a named template: {len(unmatched)}")
    print("\n".join("  " + u for u in unmatched))
    print(f"backticked table rows: {rows}; in no template: {len(rows_out)}")
    print("\n".join("  " + u for u in rows_out))


def caught(repo):
    """Each finding of a round record at a release tag v0.15.0 or later whose
    text before its first fence matches KEY, placed in a kind by the parts its
    `location:` line names. A keyword sample: a drift worded otherwise is not
    listed, and a listed one is read before it counts."""
    key = re.compile(r"drift|restat|stale copy|second copy|copy of|copies|a copy|disagree|"
                     r"contradict|out of step|no longer (says|matches|states)", re.I)
    path = re.compile(r"`?([\w./<>*-]+\.(?:md|py|yml|json|toml))")
    process = ("docs/branch-and-release.md", "docs/release-checklist.md", "docs/issues-and-milestones.md")

    def git(*a):
        return subprocess.run(["git", "-C", repo, *a], capture_output=True, text=True, check=True).stdout

    def part(p):
        if p in PARTS["P5"] or p == "README.ko.md":
            return "P5"
        for pre, name in (("docs/", "P1"), ("skills/", "P2"), ("agents/", "P3"), ("templates/", "P4"),
                          ("seal/threads/", "P7"), ("seal/specs/", "P7"), ("seal/", "P6"),
                          ("tests/", "P9"), ("evals/scripts/", "P8"), ("evals/", "P10"),
                          ("hooks/", "P8"), ("bin/", "P8"), (".github/", "P8")):
            if p.startswith(pre):
                return "P8" if name == "P2" and "/scripts/" in p else name
        return None

    def kinds(paths):
        ps, out = {part(p) for p in paths} - {None}, set()
        rule = [p for p in set(paths) if part(p) in ("P1", "P2", "P3")]
        if {"README.md", "README.ko.md"} <= set(paths) or \
                {"docs/one-root-by-lifetime.md", "docs/one-root-by-lifetime.ko.md"} <= set(paths):
            out.add("K4")
        if any(p in process for p in paths) and ps & {"P2", "P3", "P4", "P8"}:
            out.add("K9")
        if ps & {"P9", "P10"} and ps & {"P1", "P2", "P3"}:
            out.add("K5")
        if "P4" in ps and ps & {"P1", "P2", "P3", "P6"}:
            out.add("K6")
        if "P5" in ps and (ps & {"P1", "P2", "P3", "P4", "P6"} or len([p for p in set(paths) if part(p) == "P5"]) > 1):
            out.add("K3")
        if len(rule) >= 2:
            out.add("K2" if ps & {"P2", "P3"} else "K1")
        return out

    seen = {}
    tags = [t for t in git("tag", "--sort=creatordate").split() if re.fullmatch(r"v0\.(1[5-9]|2\d)\.\d+", t)]
    for tag in tags:
        for rel in git("ls-tree", "-r", "--name-only", tag, "--", "seal/").split("\n"):
            if not (rel.endswith(".md") and "round" in rel.rsplit("/", 1)[-1]):
                continue
            for block in re.split(r"(?m)^### ", git("show", f"{tag}:{rel}"))[1:]:
                head = block.split("\n", 1)[0].strip()
                loc = re.search(r"(?m)^location:(.*)$", block)
                if (rel, head) in seen or not loc or not key.search(block.split("\n```", 1)[0]):
                    continue
                ks = kinds(path.findall(loc.group(1)))
                if ks:
                    seen[(rel, head)] = (tag, sorted(ks))
    for (rel, head), (tag, ks) in sorted(seen.items(), key=lambda kv: (kv[1][1], kv[1][0], kv[0])):
        print(f"{','.join(ks)} {tag} {rel} :: {head[:110]}")
    print(f"findings {len(seen)}; " + "; ".join(
        f"{k} {sum(1 for v in seen.values() if k in v[1])}" for k in ("K1", "K2", "K3", "K4", "K5", "K6", "K9")))
